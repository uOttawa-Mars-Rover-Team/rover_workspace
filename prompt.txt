GetAprilTagPose::GetAprilTagPose(
  const std::string & name,
  const BT::NodeConfiguration & config)
: BT::SyncActionNode(name, config)
{
  // Get the node from blackboard
  auto node_ptr = config.blackboard->get<rclcpp::Node::SharedPtr>("node");
  if (node_ptr) {
    node_ = node_ptr;  // node_ptr is already a SharedPtr, not an optional
    RCLCPP_INFO(node_->get_logger(), "Using existing ROS node from blackboard");
  } else {
    node_ = rclcpp::Node::make_shared("get_apriltag_pose_node");
    RCLCPP_WARN(node_->get_logger(), "Created new ROS node - this may cause issues!");
  }
  
  // Create TF buffer and listener
  tf_buffer_ = std::make_shared<tf2_ros::Buffer>(node_->get_clock());
  tf_listener_ = std::make_shared<tf2_ros::TransformListener>(*tf_buffer_);
  
  RCLCPP_INFO(node_->get_logger(), "GetAprilTagPose node initialized successfully");
}

BT::NodeStatus GetAprilTagPose::tick()
{
  RCLCPP_INFO(node_->get_logger(), "HELOOOO!!!!!!");

  using namespace std::chrono_literals;

  // Get inputs with defaults
  std::string tag_frame_id = "base";
  double goal_offset_m = 1.0;
  std::string map_frame_id = "map";

  getInput("tag_frame_id", tag_frame_id);
  getInput("goal_offset_m", goal_offset_m);
  getInput("map_frame_id", map_frame_id);

  RCLCPP_INFO(node_->get_logger(), 
              "Looking for transform from '%s' to '%s'", 
              map_frame_id.c_str(), tag_frame_id.c_str());

  geometry_msgs::msg::PoseStamped goal_pose;
  goal_pose.header.frame_id = map_frame_id;
  goal_pose.header.stamp = node_->now();

  try {
    // Check if transform is available first
    if (!tf_buffer_->canTransform(map_frame_id, tag_frame_id, tf2::TimePointZero, 100ms)) {
      RCLCPP_WARN_THROTTLE(node_->get_logger(), *node_->get_clock(), 1000,
                           "Transform from '%s' to '%s' not available",
                           map_frame_id.c_str(), tag_frame_id.c_str());
      return BT::NodeStatus::FAILURE;
    }

    auto tag_in_map_transform = tf_buffer_->lookupTransform(
      map_frame_id, tag_frame_id, tf2::TimePointZero);
    
    double tag_x = tag_in_map_transform.transform.translation.x;
    double tag_y = tag_in_map_transform.transform.translation.y;
    double tag_yaw = tf2::getYaw(tag_in_map_transform.transform.rotation);
    
    RCLCPP_INFO(node_->get_logger(), 
                "AprilTag found at position (%.2f, %.2f) with yaw %.2f in frame '%s'",
                tag_x, tag_y, tag_yaw, map_frame_id.c_str());

    // Calculate goal position (in front of the tag)
    goal_pose.pose.position.x = tag_x - goal_offset_m * cos(tag_yaw);
    goal_pose.pose.position.y = tag_y - goal_offset_m * sin(tag_yaw);
    goal_pose.pose.position.z = 0.0;

    // Face towards the tag
    double goal_yaw = atan2(tag_y - goal_pose.pose.position.y, 
                            tag_x - goal_pose.pose.position.x);

    tf2::Quaternion q;
    q.setRPY(0, 0, goal_yaw);
    goal_pose.pose.orientation = tf2::toMsg(q);
    
    RCLCPP_INFO(node_->get_logger(), 
                "Setting navigation goal at (%.2f, %.2f) with yaw %.2f",
                goal_pose.pose.position.x, 
                goal_pose.pose.position.y,
                goal_yaw);

    setOutput("goal", goal_pose);
    
    return BT::NodeStatus::SUCCESS;
    
  } catch (const tf2::TransformException & ex) {
    RCLCPP_ERROR(node_->get_logger(), 
                 "Failed to get transform: %s", ex.what());
    return BT::NodeStatus::FAILURE;
  }
}
